#include <stdio.h>

#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "Seven_Segment.h"
#include "Scankey.h"


// ============================================================
// KEY DEFINITIONS
// ============================================================

#define B       7
#define C       8
#define PLUS    9


// ============================================================
// GLOBAL VARIABLES
// ============================================================

int num1 = 0;
int num2 = 0;

// state:
//
// 0 = entering first number
// 1 = entering second number
// 2 = showing result
//
int state = 0;


// ============================================================
// 7-SEGMENT REFRESH
//
// The display is multiplexed.
//
// We display ONE digit at a time and continuously refresh it.
//
// digit 0 = rightmost
// digit 1 = second from right
// digit 2 = third from right
// digit 3 = leftmost
//
// Example:
//
// 123:
//
// [OFF] [1] [2] [3]
//   3     2   1   0
//
// ============================================================

void Refresh_7seg(int value)
{
    static int digit = 0;

    int d0;
    int d1;
    int d2;
    int d3;


    // --------------------------------------------------------
    // Separate the number into digits
    // --------------------------------------------------------

    d0 = value % 10;
    d1 = (value / 10) % 10;
    d2 = (value / 100) % 10;
    d3 = (value / 1000) % 10;


    // --------------------------------------------------------
    // Turn off previous digit
    // --------------------------------------------------------

    CloseSevenSegment();


    // --------------------------------------------------------
    // Digit 0 = rightmost
    // --------------------------------------------------------

    if (digit == 0)
    {
        if (value != 0)
        {
            ShowSevenSegment(0, d0);
        }
    }


    // --------------------------------------------------------
    // Digit 1
    // --------------------------------------------------------

    else if (digit == 1)
    {
        if (value >= 10)
        {
            ShowSevenSegment(1, d1);
        }
    }


    // --------------------------------------------------------
    // Digit 2
    // --------------------------------------------------------

    else if (digit == 2)
    {
        if (value >= 100)
        {
            ShowSevenSegment(2, d2);
        }
    }


    // --------------------------------------------------------
    // Digit 3
    // --------------------------------------------------------

    else if (digit == 3)
    {
        if (value >= 1000)
        {
            ShowSevenSegment(3, d3);
        }
    }


    // --------------------------------------------------------
    // Short delay
    //
    // Short refresh time prevents visible flickering.
    // --------------------------------------------------------

    CLK_SysTickDelay(1000);


    // --------------------------------------------------------
    // Move to next digit
    // --------------------------------------------------------

    digit++;

    if (digit >= 4)
    {
        digit = 0;
    }
}


// ============================================================
// GET CURRENT DISPLAY NUMBER
//
// Depending on the current state:
//
// state 0 -> num1
// state 1 -> num2
// state 2 -> num1 + num2
// ============================================================

int GetDisplayNumber(void)
{
    if (state == 0)
    {
        return num1;
    }

    else if (state == 1)
    {
        return num2;
    }

    else
    {
        return num1 + num2;
    }
}


// ============================================================
// ADD DIGIT
//
// Input:
//
// 1
// 12
// 123
// 123 + 4
//
// We only keep the latest 3 digits:
//
// 123 -> 234
//
// ============================================================

void AddDigit(int *num, int digit)
{
    *num = (*num * 10 + digit) % 1000;
}


// ============================================================
// BACKSPACE
//
// 123 -> 12
// 12  -> 1
// 1   -> 0
// ============================================================

void Backspace(int *num)
{
    *num = *num / 10;
}


// ============================================================
// CLEAR EVERYTHING
// ============================================================

void ClearAll(void)
{
    num1 = 0;
    num2 = 0;

    state = 0;

    CloseSevenSegment();
}


// ============================================================
// MAIN
// ============================================================

int main(void)
{
    int key;
    int lastKey = 0;


    // ========================================================
    // SYSTEM INITIALIZATION
    // ========================================================

    SYS_Init();

    OpenSevenSegment();
    OpenKeyPad();


    // ========================================================
    // INITIAL STATE
    // ========================================================

    num1 = 0;
    num2 = 0;

    state = 0;

    CloseSevenSegment();


    // ========================================================
    // MAIN LOOP
    // ========================================================

    while (1)
    {
        // ----------------------------------------------------
        // Read keypad
        // ----------------------------------------------------

        key = ScanKey();


        // ====================================================
        // NUMBER KEYS 1~6
        //
        // IMPORTANT:
        //
        // We only execute when the key is RELEASED.
        //
        // Example:
        //
        // Press 1
        // Hold 1
        // Release 1
        //       ?
        // Add 1 ONCE
        // ====================================================

        if (lastKey >= 1 && lastKey <= 6 && key == 0)
        {
            if (state == 0)
            {
                AddDigit(&num1, lastKey);
            }

            else if (state == 1)
            {
                AddDigit(&num2, lastKey);
            }

            // If state == 2:
            // Do nothing.
            //
            // After the result, only C works.
        }


        // ====================================================
        // B KEY RELEASE
        //
        // B = Backspace
        // ====================================================

        if (lastKey == B && key == 0)
        {
            if (state == 0)
            {
                Backspace(&num1);
            }

            else if (state == 1)
            {
                Backspace(&num2);
            }

            // state == 2:
            // Do nothing.
        }


        // ====================================================
        // C KEY RELEASE
        //
        // C = CLEAR
        // ====================================================

        if (lastKey == C && key == 0)
        {
            ClearAll();
        }


        // ====================================================
        // PLUS KEY RELEASE
        // ====================================================

        if (lastKey == PLUS && key == 0)
        {
            // ------------------------------------------------
            // First +
            //
            // Save first number
            // Clear display
            // Start entering second number
            // ------------------------------------------------

            if (state == 0)
            {
                state = 1;

                num2 = 0;
            }


            // ------------------------------------------------
            // Second +
            //
            // Calculate result
            // ------------------------------------------------

            else if (state == 1)
            {
                state = 2;
            }


            // ------------------------------------------------
            // state == 2
            //
            // Do nothing.
            // Only C can clear the result.
            // ------------------------------------------------
        }


        // ====================================================
        // CONTINUOUS 7-SEGMENT REFRESH
        //
        // This happens every loop.
        //
        // We NEVER clear the display just because the number
        // changed.
        //
        // This is what keeps the display stable.
        // ====================================================

        Refresh_7seg(GetDisplayNumber());


        // ====================================================
        // SAVE CURRENT KEY
        //
        // This lets us detect:
        //
        //       key pressed
        //           ?
        //       key released
        //           ?
        //       execute once
        // ====================================================

        lastKey = key;
    }
}
