#include <stdio.h>
#include <stdlib.h>

#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"
#include "Seven_Segment.h"
#include "Scankey.h"


// ============================================================
// Key definitions
// ============================================================

#define C   8
#define R   9


// ============================================================
// RGB LED
//
// PA12 = Blue
// PA13 = Green
// PA14 = Red
//
// Active LOW:
// 0 = ON
// 1 = OFF
// ============================================================

#define RGB_BLUE    PA12
#define RGB_GREEN   PA13
#define RGB_RED     PA14


// ============================================================
// Display 1~99
//
// Only rightmost two 7-segment displays are used.
//
// Example:
//
//      7  ->  ___ ___ ___  7
//     12  ->  ___ ___  1  2
//     37  ->  ___ ___  3  7
//     99  ->  ___ ___  9  9
//
// ShowSevenSegment(0, x)
//      -> rightmost digit
//
// ShowSevenSegment(1, x)
//      -> second from right
// ============================================================

void Display_7seg(uint16_t value)
{
    uint8_t tens;
    uint8_t ones;

    tens = value / 10;
    ones = value % 10;


    // --------------------------------------------------------
    // Tens
    // --------------------------------------------------------

    if (tens != 0)
    {
        CloseSevenSegment();

        ShowSevenSegment(1, tens);

        CLK_SysTickDelay(5000);
    }


    // --------------------------------------------------------
    // Ones
    // --------------------------------------------------------

    CloseSevenSegment();

    ShowSevenSegment(0, ones);

    CLK_SysTickDelay(5000);
}


// ============================================================
// Clear 7-segment
// ============================================================

void Clear_7seg(void)
{
    CloseSevenSegment();

    CLK_SysTickDelay(5000);
}


// ============================================================
// Check prime number
//
// Return:
//      1 = prime
//      0 = not prime
// ============================================================

int IsPrime(int n)
{
    int i;


    // 0 and 1 are not prime

    if (n < 2)
    {
        return 0;
    }


    // Try dividing by 2, 3, 4...
    // Only need to check up to sqrt(n)

    for (i = 2; i * i <= n; i++)
    {
        if (n % i == 0)
        {
            return 0;
        }
    }


    return 1;
}


// ============================================================
// RGB OFF
// ============================================================

void RGB_Clear(void)
{
    RGB_RED   = 1;
    RGB_GREEN = 1;
    RGB_BLUE  = 1;
}


// ============================================================
// Decide RGB color
//
// Prime       -> RED
// Multiple 2  -> BLUE
// Multiple 3  -> GREEN
//
// IMPORTANT:
// These MUST be separate if statements.
//
// This allows mixed colors.
// ============================================================

void Update_RGB(int num)
{
    // First turn everything OFF

    RGB_Clear();


    // --------------------------------------------------------
    // Prime -> RED
    // --------------------------------------------------------

    if (IsPrime(num))
    {
        RGB_RED = 0;
    }


    // --------------------------------------------------------
    // Multiple of 2 -> BLUE
    // --------------------------------------------------------

    if (num % 2 == 0)
    {
        RGB_BLUE = 0;
    }


    // --------------------------------------------------------
    // Multiple of 3 -> GREEN
    // --------------------------------------------------------

    if (num % 3 == 0)
    {
        RGB_GREEN = 0;
    }
}


// ============================================================
// Buzzer
//
// IMPORTANT:
//
// While the buzzer is working,
// keep refreshing the 7-segment.
//
// Otherwise the number may disappear / flicker.
//
// number = number of beeps
// displayNum = number currently displayed
// ============================================================

void Buzz(int number, int displayNum)
{
    int i;

    for (i = 0; i < number; i++)
    {
        // ----------------------------------------------------
        // Buzzer ON
        // ----------------------------------------------------

        PB11 = 0;

        Display_7seg(displayNum);


        // ----------------------------------------------------
        // Buzzer OFF
        // ----------------------------------------------------

        PB11 = 1;

        Display_7seg(displayNum);
    }
}


// ============================================================
// MAIN
// ============================================================

int main(void)
{
    int key;

    // Remember previous key
    int lastKey = 0;

    // Current random number
    int num = 0;


    // ========================================================
    // System initialization
    // ========================================================

    SYS_Init();

    OpenSevenSegment();

    OpenKeyPad();


    // ========================================================
    // Buzzer
    // ========================================================

    GPIO_SetMode(PB, BIT11, GPIO_MODE_OUTPUT);

    // Buzzer OFF

    PB11 = 1;


    // ========================================================
    // RGB LED
    // ========================================================

    GPIO_SetMode(PA, BIT12, GPIO_MODE_OUTPUT);

    GPIO_SetMode(PA, BIT13, GPIO_MODE_OUTPUT);

    GPIO_SetMode(PA, BIT14, GPIO_MODE_OUTPUT);


    // RGB OFF

    RGB_Clear();


    // ========================================================
    // Random seed
    // ========================================================

    srand(SysTick->VAL);


    // ========================================================
    // Initial state
    // ========================================================

    Clear_7seg();

    RGB_Clear();

    PB11 = 1;


    // ========================================================
    // MAIN LOOP
    // ========================================================

    while (1)
    {
        // ----------------------------------------------------
        // Read keypad
        // ----------------------------------------------------

        key = ScanKey();


        // ====================================================
        // R RELEASE DETECTION
        //
        // Previous state:
        //
        //      R
        //
        // Current state:
        //
        //      0
        //
        // Means:
        //
        //      R was pressed
        //      ?
        //      R was released
        //      ?
        //      execute ONCE
        // ====================================================

        if (lastKey == R && key == 0)
        {
            // ------------------------------------------------
            // Generate ONE random number
            //
            // 1 ~ 99
            // ------------------------------------------------

            num = rand() % 99 + 1;


            // ------------------------------------------------
            // Immediately update RGB
            // ------------------------------------------------

            Update_RGB(num);


            // ------------------------------------------------
            // Immediately display number
            //
            // This happens BEFORE the buzzer.
            //
            // Therefore there is no "one beat delay".
            // ------------------------------------------------

            Display_7seg(num);


            // ------------------------------------------------
            // Prime -> beep once
            // ------------------------------------------------

            if (IsPrime(num))
            {
                Buzz(1, num);
            }
        }


        // ====================================================
        // C RELEASE DETECTION
        // ====================================================

        if (lastKey == C && key == 0)
        {
            // Clear number

            num = 0;


            // 7-segment OFF

            Clear_7seg();


            // RGB OFF

            RGB_Clear();


            // Buzzer OFF

            PB11 = 1;
        }


        // ====================================================
        // KEEP DISPLAYING CURRENT NUMBER
        //
        // IMPORTANT:
        //
        // We DO NOT call rand() here.
        //
        // Therefore:
        //
        //      num = 37
        //
        // will stay:
        //
        //      37
        //      37
        //      37
        //      37
        //      ...
        // ====================================================

        if (num != 0)
        {
            Display_7seg(num);
        }


        // ====================================================
        // Remember current key
        //
        // This is what allows us to detect:
        //
        //      R -> release
        //      C -> release
        // ====================================================

        lastKey = key;
    }
}
