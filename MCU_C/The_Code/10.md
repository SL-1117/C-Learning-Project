#include <stdio.h>

#include "NUC100Series.h"
#include "MCU_init.h"
#include "SYS_init.h"

// ============================================================
// RUNNING LED
//
// PC12 -> PC13 -> PC14 -> PC15
//
// led_position:
// 0 = PC12
// 1 = PC13
// 2 = PC14
// 3 = PC15
// ============================================================

volatile int led_position = 0;

// ============================================================
// DIRECTION
//
// 0 = RIGHT
// 1 = LEFT
// ============================================================

volatile int direction = 0;

// ============================================================
// 1 SECOND TIMER
//
// SysTick runs every 1 ms.
//
// 1000 ms = 1 second
// ============================================================

volatile uint32_t led_timer = 0;

// ============================================================
// RGB LED
//
// PA12 = RED
// PA13 = GREEN
// PA14 = BLUE
//
// Active LOW
//
// red   = 1 -> ON
// red   = 0 -> OFF
//
// green = 1 -> ON
// green = 0 -> OFF
//
// blue  = 1 -> ON
// blue  = 0 -> OFF
// ============================================================

volatile int red = 0;
volatile int green = 0;
volatile int blue = 0;

// ============================================================
// KEYPAD
//
//       1   2   3
//       4   5   6
//       7   8   9
//
// PA3~PA5 = scanning lines
// PA0~PA2 = reading lines
// ============================================================

volatile int pressed_key = 0;

// ============================================================
// RGB UPDATE
// ============================================================

void RGB_Update(void)
{
// RED
if (red)
PA12 = 0;
else
PA12 = 1;

// GREEN
if (green)
    PA13 = 0;
else
    PA13 = 1;


// BLUE
if (blue)
    PA14 = 0;
else
    PA14 = 1;

}

// ============================================================
// RGB INITIALIZATION
// ============================================================

void Init_RGB(void)
{
GPIO_SetMode(PA, BIT12, GPIO_MODE_OUTPUT);
GPIO_SetMode(PA, BIT13, GPIO_MODE_OUTPUT);
GPIO_SetMode(PA, BIT14, GPIO_MODE_OUTPUT);

// RGB OFF

PA12 = 1;
PA13 = 1;
PA14 = 1;

}

// ============================================================
// RUNNING LED INITIALIZATION
// ============================================================

void Init_LED(void)
{
GPIO_SetMode(PC, BIT12, GPIO_MODE_OUTPUT);
GPIO_SetMode(PC, BIT13, GPIO_MODE_OUTPUT);
GPIO_SetMode(PC, BIT14, GPIO_MODE_OUTPUT);
GPIO_SetMode(PC, BIT15, GPIO_MODE_OUTPUT);

// All LEDs OFF

PC12 = 1;
PC13 = 1;
PC14 = 1;
PC15 = 1;

}

// ============================================================
// SHOW CURRENT RUNNING LED
//
// 0 -> PC12
// 1 -> PC13
// 2 -> PC14
// 3 -> PC15
// ============================================================

void Show_Running_LED(void)
{
// Turn ALL LEDs OFF first

PC12 = 1;
PC13 = 1;
PC14 = 1;
PC15 = 1;


// Turn ON current LED

if (led_position == 0)
{
    PC12 = 0;
}

else if (led_position == 1)
{
    PC13 = 0;
}

else if (led_position == 2)
{
    PC14 = 0;
}

else if (led_position == 3)
{
    PC15 = 0;
}

}

// ============================================================
// MOVE ONE LED
//
// RIGHT:
//
// PC12 -> PC13 -> PC14 -> PC15 -> PC12
//
// LEFT:
//
// PC12 -> PC15 -> PC14 -> PC13 -> PC12
// ============================================================

void Move_Running_LED(void)
{
if (direction == 0)
{
// ====================================================
// RIGHT
// ====================================================

    if (led_position == 3)
    {
        led_position = 0;
    }
    else
    {
        led_position++;
    }
}

else
{
    // ====================================================
    // LEFT
    // ====================================================

    if (led_position == 0)
    {
        led_position = 3;
    }
    else
    {
        led_position--;
    }
}


// Show new position immediately

Show_Running_LED();

}

// ============================================================
// SYSTICK INTERRUPT
//
// SysTick = 1 ms
//
// 1000 times = 1 second
//
// Every 1 second:
//     move LED ONE position
// ============================================================

void SysTick_Handler(void)
{
led_timer++;

if (led_timer >= 1000)
{
    led_timer = 0;

    Move_Running_LED();
}

}

// ============================================================
// KEYPAD INITIALIZATION
//
//       1   2   3
//       4   5   6
//       7   8   9
//
// PA3~PA5 = scanning
// PA0~PA2 = reading
// ============================================================

void Init_Keypad(void)
{
// Reading lines

GPIO_SetMode(PA, BIT0, GPIO_MODE_QUASI);
GPIO_SetMode(PA, BIT1, GPIO_MODE_QUASI);
GPIO_SetMode(PA, BIT2, GPIO_MODE_QUASI);


// Scanning lines

GPIO_SetMode(PA, BIT3, GPIO_MODE_QUASI);
GPIO_SetMode(PA, BIT4, GPIO_MODE_QUASI);
GPIO_SetMode(PA, BIT5, GPIO_MODE_QUASI);


// ========================================================
// IDLE STATE
// ========================================================

PA3 = 0;
PA4 = 0;
PA5 = 0;

PA0 = 1;
PA1 = 1;
PA2 = 1;


// ========================================================
// INTERRUPT ON PA0~PA2
// ========================================================

GPIO_EnableInt(
    PA,
    0,
    GPIO_INT_BOTH_EDGE
);

GPIO_EnableInt(
    PA,
    1,
    GPIO_INT_BOTH_EDGE
);

GPIO_EnableInt(
    PA,
    2,
    GPIO_INT_BOTH_EDGE
);


NVIC_EnableIRQ(GPAB_IRQn);


// ========================================================
// DEBOUNCE
// ========================================================

GPIO_SET_DEBOUNCE_TIME(
    GPIO_DBCLKSRC_LIRC,
    GPIO_DBCLKSEL_64
);

GPIO_ENABLE_DEBOUNCE(
    PA,
    BIT0 | BIT1 | BIT2
);

}

// ============================================================
// GET KEY
//
// PA3 LOW:
//
// 1 2 3
//
// PA4 LOW:
//
// 4 5 6
//
// PA5 LOW:
//
// 7 8 9
// ============================================================

int GetKey(void)
{
int key = 0;

// ========================================================
// ROW 1
// 1 2 3
// ========================================================

PA3 = 0;
PA4 = 1;
PA5 = 1;


if (PA0 == 0)
{
    key = 1;
}

else if (PA1 == 0)
{
    key = 2;
}

else if (PA2 == 0)
{
    key = 3;
}


// ========================================================
// ROW 2
// 4 5 6
// ========================================================

if (key == 0)
{
    PA3 = 1;
    PA4 = 0;
    PA5 = 1;


    if (PA0 == 0)
    {
        key = 4;
    }

    else if (PA1 == 0)
    {
        key = 5;
    }

    else if (PA2 == 0)
    {
        key = 6;
    }
}


// ========================================================
// ROW 3
// 7 8 9
// ========================================================

if (key == 0)
{
    PA3 = 1;
    PA4 = 1;
    PA5 = 0;


    if (PA0 == 0)
    {
        key = 7;
    }

    else if (PA1 == 0)
    {
        key = 8;
    }

    else if (PA2 == 0)
    {
        key = 9;
    }
}


// ========================================================
// RETURN TO IDLE
// ========================================================

PA3 = 0;
PA4 = 0;
PA5 = 0;

PA0 = 1;
PA1 = 1;
PA2 = 1;


return key;

}

// ============================================================
// KEYPAD INTERRUPT
//
// PRESS:
//     Remember the key.
//
// RELEASE:
//     Execute the action.
//
// 1 = RED
// 2 = GREEN
// 3 = BLUE
//
// IMPORTANT:
// No OpenKeyPad()
// No ScanKey()
// ============================================================

void GPAB_IRQHandler(void)
{
int key;

// ========================================================
// CLEAR INTERRUPT FLAGS
// ========================================================

if (PA->ISRC & BIT0)
{
    PA->ISRC = BIT0;
}

if (PA->ISRC & BIT1)
{
    PA->ISRC = BIT1;
}

if (PA->ISRC & BIT2)
{
    PA->ISRC = BIT2;
}


// ========================================================
// GET KEY
// ========================================================

key = GetKey();


// ========================================================
// PRESS
//
// Remember key only.
// ========================================================

if (key != 0)
{
    pressed_key = key;
}


// ========================================================
// RELEASE
//
// Execute action.
// ========================================================

else
{
    if (pressed_key == 1)
    {
        red = !red;
    }

    else if (pressed_key == 2)
    {
        green = !green;
    }

    else if (pressed_key == 3)
    {
        blue = !blue;
    }


    // Update RGB

    RGB_Update();


    // Clear previous key

    pressed_key = 0;
}

}

// ============================================================
// INT1 INTERRUPT
//
// PB15
//
// RIGHT <-> LEFT
// ============================================================

void EINT1_IRQHandler(void)
{
// Clear interrupt flag

GPIO_CLR_INT_FLAG(PB, BIT15);


// Change direction

if (direction == 0)
{
    direction = 1;
}
else
{
    direction = 0;
}

}

// ============================================================
// INT1 INITIALIZATION
// ============================================================

void Init_INT1(void)
{
GPIO_SetMode(
PB,
BIT15,
GPIO_MODE_INPUT
);

// INT1 on PB15

GPIO_EnableEINT1(
    PB,
    15,
    GPIO_INT_RISING
);


NVIC_EnableIRQ(EINT1_IRQn);


// ========================================================
// DEBOUNCE
// ========================================================

GPIO_SET_DEBOUNCE_TIME(
    GPIO_DBCLKSRC_LIRC,
    GPIO_DBCLKSEL_64
);

GPIO_ENABLE_DEBOUNCE(
    PB,
    BIT15
);

}

// ============================================================
// MAIN
// ============================================================

int main(void)
{
// ========================================================
// SYSTEM
// ========================================================

SYS_Init();


// ========================================================
// INITIALIZE HARDWARE
// ========================================================

Init_LED();

Init_RGB();

Init_Keypad();

Init_INT1();


// ========================================================
// INITIAL STATE
// ========================================================

// Start RIGHT

direction = 0;


// Start at PC12

led_position = 0;


// Timer starts from 0

led_timer = 0;


// Show first LED

Show_Running_LED();


// ========================================================
// SYSTICK
//
// SystemCoreClock / 1000
//
// = 1 interrupt every 1 ms
// ========================================================

SysTick_Config(SystemCoreClock / 1000);


// ========================================================
// ENABLE GLOBAL INTERRUPTS
// ========================================================

__enable_irq();


// ========================================================
// MAIN LOOP
// ========================================================

while (1)
{
    // Nothing needed here.

    // Running LED:
    // controlled by SysTick

    // Keypad:
    // controlled by GPAB interrupt

    // INT1:
    // controlled by EINT1 interrupt
}

}
